cmake_minimum_required(VERSION 3.5)
project(galbot_s1_web)

if("$ENV{ROS_VERSION}" STREQUAL "2")
  find_package(ament_cmake REQUIRED)
  install(DIRECTORY web launch DESTINATION share/${PROJECT_NAME})
  install(PROGRAMS scripts/web_server.py DESTINATION lib/${PROJECT_NAME})
  ament_package()
else()
  find_package(catkin REQUIRED)
  catkin_package()
  install(DIRECTORY web launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
  catkin_install_python(PROGRAMS scripts/web_server.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
endif()
